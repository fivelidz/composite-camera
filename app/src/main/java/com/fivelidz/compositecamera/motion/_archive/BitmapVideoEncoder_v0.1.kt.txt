package com.fivelidz.compositecamera.motion

import android.graphics.Bitmap
import android.graphics.Canvas
import android.graphics.Rect
import android.media.MediaCodec
import android.media.MediaCodecInfo
import android.media.MediaFormat
import android.media.MediaMuxer
import android.util.Log
import android.view.Surface
import java.io.File
import java.nio.ByteBuffer

/**
 * Encodes a stream of Bitmaps to an MP4 file via MediaCodec H.264.
 * Frames must be the SAME dimensions as the encoder was constructed with.
 *
 * Usage:
 *   val enc = BitmapVideoEncoder(file, w, h, fps = 30)
 *   enc.start()
 *   for (frame in frames) enc.encode(frame, timestampNs)
 *   val out = enc.stop()
 */
class BitmapVideoEncoder(
    private val file: File,
    width: Int,
    height: Int,
    private val fps: Int = 30,
    private val bitrate: Int = 6_000_000,
) {
    private val TAG = "BitmapVideoEncoder"
    // MediaCodec requires width/height divisible by 2.
    private val w = width and 1.inv()
    private val h = height and 1.inv()

    private var codec: MediaCodec? = null
    private var muxer: MediaMuxer? = null
    private var inputSurface: Surface? = null
    private var trackIndex = -1
    private var muxerStarted = false
    private var firstTs: Long = -1L
    private var frameCount = 0

    fun start() {
        val format = MediaFormat.createVideoFormat(MediaFormat.MIMETYPE_VIDEO_AVC, w, h).apply {
            setInteger(MediaFormat.KEY_COLOR_FORMAT,
                MediaCodecInfo.CodecCapabilities.COLOR_FormatSurface)
            setInteger(MediaFormat.KEY_BIT_RATE, bitrate)
            setInteger(MediaFormat.KEY_FRAME_RATE, fps)
            setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 2)
        }
        codec = MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_VIDEO_AVC).apply {
            configure(format, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
            inputSurface = createInputSurface()
            start()
        }
        muxer = MediaMuxer(file.absolutePath, MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4)
    }

    fun encode(bitmap: Bitmap, ts: Long) {
        val codec = this.codec ?: return
        val surface = this.inputSurface ?: return

        if (firstTs < 0) firstTs = ts

        val canvas = surface.lockCanvas(null)
        try {
            val dst = Rect(0, 0, w, h)
            canvas.drawBitmap(bitmap, null, dst, null)
        } finally {
            surface.unlockCanvasAndPost(canvas)
        }
        frameCount++

        drainEncoder(endOfStream = false)
    }

    fun stop(): File {
        val codec = this.codec ?: return file
        try {
            codec.signalEndOfInputStream()
        } catch (t: Throwable) { Log.w(TAG, "signalEndOfInputStream: $t") }
        drainEncoder(endOfStream = true)
        try { codec.stop(); codec.release() } catch (_: Throwable) {}
        try { muxer?.stop(); muxer?.release() } catch (_: Throwable) {}
        this.codec = null
        this.muxer = null
        return file
    }

    fun release() {
        try { codec?.stop() } catch (_: Throwable) {}
        try { codec?.release() } catch (_: Throwable) {}
        try { muxer?.stop() } catch (_: Throwable) {}
        try { muxer?.release() } catch (_: Throwable) {}
        codec = null; muxer = null
    }

    private fun drainEncoder(endOfStream: Boolean) {
        val codec = this.codec ?: return
        val muxer = this.muxer ?: return
        val info = MediaCodec.BufferInfo()
        val timeoutUs = if (endOfStream) 10_000L else 0L
        while (true) {
            val outIdx = codec.dequeueOutputBuffer(info, timeoutUs)
            if (outIdx == MediaCodec.INFO_TRY_AGAIN_LATER) {
                if (!endOfStream) return
            } else if (outIdx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
                if (muxerStarted) error("Format changed twice")
                trackIndex = muxer.addTrack(codec.outputFormat)
                muxer.start()
                muxerStarted = true
            } else if (outIdx < 0) {
                // ignore
            } else {
                val buf: ByteBuffer = codec.getOutputBuffer(outIdx)
                    ?: error("Encoder output buffer was null")
                if (info.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG != 0) info.size = 0
                if (info.size != 0 && muxerStarted) {
                    buf.position(info.offset)
                    buf.limit(info.offset + info.size)
                    // Use our own monotonic timestamp (Surface input ignores BufferInfo.presentationTimeUs
                    // for muxing — but we still need it set for the muxer call).
                    val ptsUs = (frameCount.toLong() * 1_000_000L / fps)
                    info.presentationTimeUs = ptsUs
                    muxer.writeSampleData(trackIndex, buf, info)
                }
                codec.releaseOutputBuffer(outIdx, false)
                if (info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM != 0) return
            }
        }
    }
}
